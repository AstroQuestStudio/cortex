pub fn grid_op_000(grid: &mut Grid, x: usize, y: usize, flags: Flags) -> Result<Cell, GridError>
ParameterTypeDescription
grid&mut GridThe grid to modify.
xusizeColumn, zero based.
yusizeRow, zero based.

Example​

let c = grid_op_000(&mut grid, 0, 0, Flags::empty())?;
